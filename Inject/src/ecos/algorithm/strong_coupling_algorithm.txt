#include "ecos/algorithm/strong_coupling_algorithm.hpp"

#include "ecos/logger/logger.hpp"

#include <algorithm>
#include <cmath>
#include <execution>
#include <map>

using namespace ecos;

namespace
{

struct instance_wrapper
{
    model_instance* instance;

    std::map<std::string, double> prevOutputValues;
    std::map<std::string, double> prevPrevOutputValues;
};

/**************************************************************
 * 保持器类型
 **************************************************************/
enum class HolderType
{
    None,
    ZeroOrder,
    FirstOrder,
    SecondOrder
};

} // namespace

/**************************************************************
 * strong_coupling_algorithm::impl
 **************************************************************/
class strong_coupling_algorithm::impl
{
public:
    impl(double stepSize,
        std::size_t maxIterations,
        bool parallel,
        HolderType outputHolderType,
        HolderType inputHolderType)
        : stepSize_(stepSize)
        , maxIterations_(maxIterations)
        , parallel_(parallel)
        , outputHolderType_(outputHolderType)
        , inputHolderType_(inputHolderType)
    { }

    void model_instance_added(model_instance* instance)
    {
        log::debug("Strong coupling: model added [{}]",
            instance->instanceName());
        instances_.push_back(instance_wrapper{instance});
    }

    double step(double currentTime)
    {
        log::debug("=== Step at t = {} ===", currentTime);

        /**********************************************************
         * Step 1: FMU 时间推进
         **********************************************************/
        for (auto& w : instances_) {
            w.instance->get_properties().apply_sets();
            w.instance->step(currentTime, stepSize_);
            w.instance->get_properties().apply_gets();
        }

        /**********************************************************
         * Step 2: 接口残余能量计算
         **********************************************************/
        compute_residual_energy();

        /**********************************************************
         * Step 3: 强耦合接口迭代
         **********************************************************/
        for (std::size_t iter = 0; iter < maxIterations_; ++iter) {
            log::debug("--- Interface iteration {} ---", iter);

            for (auto& w : instances_) {
                auto& props = w.instance->get_properties();

                apply_output_holders(w);
                apply_input_modifiers(w);

                props.apply_sets();
                props.apply_gets();

                print_property_values(w);
            }
        }

        return currentTime + stepSize_;
    }

private:
    /**********************************************************
     * 参数
     **********************************************************/
    double stepSize_;
    std::size_t maxIterations_;
    bool parallel_;

    HolderType outputHolderType_;
    HolderType inputHolderType_;

    std::vector<instance_wrapper> instances_;

    /**********************************************************
     * 能量修正状态
     **********************************************************/
    double lastResidualPower_ = 0.0;
    double lastResidualEnergy_ = 0.0;
    double accumulatedEnergyError_ = 0.0;

    double energyGain_ = 1; // μ
    double integralGain_ = 0; // ν

    /**********************************************************
     * 输出保持器
     **********************************************************/
    void apply_output_holders(instance_wrapper& w)
    {
        auto& props = w.instance->get_properties();

        for (const auto& [name, prop] : props.get_reals()) {
            const double currentValue = prop->get_value();

            switch (outputHolderType_) {
                case HolderType::None:
                    break;

                case HolderType::ZeroOrder: {
                    auto it = w.prevOutputValues.find(name);
                    if (it != w.prevOutputValues.end()) {
                        prop->set_output_modifier(
                            [v = it->second](double) { return v; });
                    }
                    break;
                }

                case HolderType::FirstOrder: {
                    auto it1 = w.prevOutputValues.find(name);
                    auto it2 = w.prevPrevOutputValues.find(name);

                    if (it1 != w.prevOutputValues.end() &&
                        it2 != w.prevPrevOutputValues.end()) {
                        const double yk = it1->second;
                        const double slope = it1->second - it2->second;
                        prop->set_output_modifier(
                            [yk, slope](double) {
                                return yk + slope;
                            });
                    }
                    break;
                }

                case HolderType::SecondOrder:
                    break;
            }

            w.prevPrevOutputValues[name] = w.prevOutputValues[name];
            w.prevOutputValues[name] = currentValue;
        }
    }

    /**********************************************************
     * 接口残余能量计算
     **********************************************************/
    void compute_residual_energy()
    {
        if (instances_.size() < 2)
            return;

        auto& A = instances_[0];
        auto& B = instances_[1];

        auto& propsA = A.instance->get_properties();
        auto& propsB = B.instance->get_properties();

        auto* FA = propsA.get_real_property("F");
        auto* VA = propsA.get_real_property("V");
        auto* FB = propsB.get_real_property("F");
        auto* VB = propsB.get_real_property("V");

        if (!FA || !VA || !FB || !VB)
            return;

        const double PA = FA->get_value() * VA->get_value();
        const double PB = FB->get_value() * VB->get_value();

        lastResidualPower_ = (PA - PB);
        lastResidualEnergy_ = stepSize_ * lastResidualPower_;

        accumulatedEnergyError_ = energyGain_ * lastResidualEnergy_;

        log::debug("[Energy] dP = {}, dE = {}, U = {}",
            lastResidualPower_,
            lastResidualEnergy_,
            accumulatedEnergyError_);
    }

    /**********************************************************
     * 输入修饰器（能量修正在这里注入）
     **********************************************************/
    void apply_input_modifiers(instance_wrapper& w)
    {
        auto& props = w.instance->get_properties();

        // 只对第二个子系统施加能量修正（避免过约束）
        if (w.instance != instances_[1].instance)
            return;

        auto* Fprop = props.get_real_property("F");
        auto* Vprop = props.get_real_property("V");

        if (!Fprop || !Vprop)
            return;

        const double V = Vprop->get_value();
        if (std::abs(V) < 1e-10)
            return;

        const double Fcorr =
            -(lastResidualEnergy_ + integralGain_ * accumulatedEnergyError_) /  (stepSize_ * V);

        Fprop->set_input_modifier(
            [Fcorr](const double& Fin) {
                return Fin + Fcorr;
            });

        log::debug("[EnergyCorrection] Fcorr = {}", Fcorr);
    }

    /**********************************************************
     * 调试输出
     **********************************************************/

    void print_property_values(const instance_wrapper& w)
    {
        auto& props = w.instance->get_properties();

        for (const auto& name : {"F", "V"}) {
            if (auto* p = props.get_real_property(name)) {
                log::debug("[{}] {} = {}",
                    w.instance->instanceName(),
                    name,
                    p->get_value());
            }
        }
    }
};

/**************************************************************
 * strong_coupling_algorithm 对外接口
 **************************************************************/
strong_coupling_algorithm::strong_coupling_algorithm(
    double stepSize,
    std::size_t maxIterations,
    bool parallel,
    int outputHolderType,
    int inputHolderType)
    : pimpl_(std::make_unique<impl>(
          stepSize,
          maxIterations,
          parallel,
          static_cast<HolderType>(outputHolderType),
          static_cast<HolderType>(inputHolderType)))
{ }

void strong_coupling_algorithm::model_instance_added(
    model_instance* instance)
{
    pimpl_->model_instance_added(instance);
}

double strong_coupling_algorithm::step(double currentTime)
{
    return pimpl_->step(currentTime);
}

strong_coupling_algorithm::~strong_coupling_algorithm() = default;
