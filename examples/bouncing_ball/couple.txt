
#include "ecos/simulation.hpp"
#include "ecos/structure/simulation_structure.hpp"
#include <ecos/algorithm/strong_coupling_algorithm.hpp> // ← 新增
#include <ecos/listeners/csv_writer.hpp>
#include <ecos/logger/logger.hpp>
#include <ecos/util/plotter.hpp>

using namespace ecos;

int main()
{
    set_logging_level(log::level::debug);

    std::string resultFile{"results/bouncing_ball.csv"};
    const std::string fmuPath1{std::string(DATA_FOLDER) + "/fmus/test/nodes_input.fmu"};
    const std::string fmuPath2{std::string(DATA_FOLDER) + "/fmus/test/nodes_output.fmu"};

    simulation_structure ss;
    ss.add_model("NodesInput", fmuPath1);
    ss.add_model("NodesOutput", fmuPath2);

    ss.make_connection<std::vector<double>>(
        "NodesOutput::outputArray",
        "NodesInput::inputArray");
    ss.make_connection<std::vector<double>>(
        "NodesInput::sumArray",
        "NodesOutput::inputArray");

    // ⭐ 核心修改：使用强耦合算法 ⭐
    const auto sim = ss.load(
        std::make_unique<strong_coupling_algorithm>(
            1.0, // stepSize
            10, // maxIterations
            false // parallel（false = Gauss-Seidel）
            ));

    sim->add_listener("csv_writer",
        std::make_unique<csv_writer>(resultFile));

    sim->init();
    sim->step_for(5);
    sim->terminate();
}
