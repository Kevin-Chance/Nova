
#include "nova/simulation.hpp"
#include "nova/structure/simulation_structure.hpp"
#include <nova/algorithm/fixed_step_algorithm.hpp>
#include <nova/listeners/csv_writer.hpp>
#include <nova/logger/logger.hpp>
#include <nova/util/plotter.hpp>

using namespace nova;

int main()
{

    set_logging_level(log::level::debug);

    std::string resultFile{"results/bouncing_ball.csv"};
    const std::string fmuPath1{std::string(DATA_FOLDER) + "/fmus/test/nodes_input.fmu"};
    const std::string fmuPath2{std::string(DATA_FOLDER) + "/fmus/test/nodes_output.fmu"};


    simulation_structure ss;
    ss.add_model("NodesInput", fmuPath1);
    ss.add_model("NodesOutput", fmuPath2);


    ss.make_connection<std::vector<double>>("NodesOutput::outputArray", "NodesInput::inputArray");
    ss.make_connection<std::vector<double>>("NodesInput::sumArray", "NodesOutput::inputArray");

    const auto sim = ss.load(std::make_unique<fixed_step_algorithm>(1));

    sim->add_listener("csv_writer", std::make_unique<csv_writer>(resultFile));

    sim->init();
    sim->step_for(5);
    sim->terminate();

}
